AI Ambassador

Deployment Guide

Deploy to Azure production environment

Prerequisites

  • Azure subscription with credits
  • Azure CLI installed (az --version)
  • Python 3.11 installed
  • Azure Functions Core Tools v4

Step 1: Deploy Infrastructure

# Login to Azure
az login

# Create resource group
az group create \
  --name aibast-prod-rg \
  --location eastus

# Deploy Bicep template
az deployment group create \
  --resource-group aibast-prod-rg \
  --template-file main.bicep \
  --parameters parameters/prod.parameters.json

Step 2: Create Azure OpenAI Resource

az cognitiveservices account create \
  --name aibast-openai-prod \
  --resource-group aibast-prod-rg \
  --kind OpenAI \
  --sku S0 \
  --location eastus \
  --custom-domain aibast-openai-prod

# Deploy GPT-4 model
az cognitiveservices account deployment create \
  --name aibast-openai-prod \
  --resource-group aibast-prod-rg \
  --deployment-name gpt-4 \
  --model-name gpt-4 \
  --model-version "0613" \
  --model-format OpenAI \
  --sku-capacity 10 \
  --sku-name "Standard"

Step 3: Create Storage Account

az storage account create \
  --name aibastprodstore \
  --resource-group aibast-prod-rg \
  --location eastus \
  --sku Standard_LRS

# Create file shares
az storage share create \
  --name ambassadors \
  --account-name aibastprodstore

az storage share create \
  --name users \
  --account-name aibastprodstore

az storage share create \
  --name agents \
  --account-name aibastprodstore

Step 4: Deploy Function App

cd Copilot-Agent-365-main

# Create function app
az functionapp create \
  --resource-group aibast-prod-rg \
  --consumption-plan-location eastus \
  --runtime python \
  --runtime-version 3.11 \
  --functions-version 4 \
  --name aibast-function-prod \
  --storage-account aibastprodstore

# Deploy code
func azure functionapp publish aibast-function-prod

Step 5: Configure Application Settings

az functionapp config appsettings set \
  --name aibast-function-prod \
  --resource-group aibast-prod-rg \
  --settings \
    "AZURE_OPENAI_API_KEY=@Microsoft.KeyVault(SecretUri=https://your-keyvault.vault.azure.net/secrets/openai-key/)" \
    "AZURE_OPENAI_ENDPOINT=https://aibast-openai-prod.openai.azure.com/" \
    "AZURE_OPENAI_DEPLOYMENT_NAME=gpt-4" \
    "AZURE_FILES_SHARE_NAME=ambassadors" \
    "ASSISTANT_NAME=AI Ambassador"

Step 6: Setup Custom Domain

# Add custom domain
az functionapp config hostname add \
  --webapp-name aibast-function-prod \
  --resource-group aibast-prod-rg \
  --hostname ai-ambassadors.app

# Enable HTTPS
az functionapp config ssl bind \
  --name aibast-function-prod \
  --resource-group aibast-prod-rg \
  --certificate-thumbprint {thumbprint} \
  --ssl-type SNI

CI/CD Setup with GitHub Actions

name: Deploy to Azure

on:
  push:
    branches: [main]

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2
      
      - name: Setup Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.11'
      
      - name: Install dependencies
        run: |
          cd Copilot-Agent-365-main
          pip install -r requirements.txt
      
      - name: Deploy to Azure
        uses: Azure/functions-action@v1
        with:
          app-name: aibast-function-prod
          package: Copilot-Agent-365-main
          publish-profile: ${{ secrets.AZURE_FUNCTIONAPP_PUBLISH_PROFILE }}
